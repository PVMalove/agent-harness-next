      target.classList.add("is-target");
      window.setTimeout(() => target.classList.remove("is-target"), 1600);
      history.replaceState(null, "", location.pathname + location.search + "#" + id);
    }

    chips.filter((chip) => chip.dataset.category).forEach((chip) => {
      chip.addEventListener("click", () => {
        setCategory(chip.dataset.category);
        apply();
        syncUrl();
      });
    });

    sortButton.addEventListener("click", () => {
      setAlpha(!alpha);
      apply();
      syncUrl();
    });

    search.addEventListener("input", () => {
      query = search.value;
      apply();
      syncUrl();
    });

    document.addEventListener("keydown", (event) => {
      const target = event.target;
      const typing = target instanceof HTMLElement && (
        target.tagName === "INPUT" || target.tagName === "TEXTAREA" ||
        target.tagName === "SELECT" || target.isContentEditable
      );
      if (event.key === "/" && !typing) {
        event.preventDefault();
        event.stopImmediatePropagation();
        search.focus();
      }
      if (event.key === "Escape" && document.activeElement === search) {
        search.value = "";
        query = "";
        search.blur();
        apply();
        syncUrl();
      }
    }, true);

    main.addEventListener("click", (event) => {
      const copy = event.target.closest(".ah-term-copy");
      if (copy) {
        event.preventDefault();
        navigator.clipboard.writeText(copy.dataset.copy).then(() => {
          copy.classList.add("is-copied");
          window.setTimeout(() => copy.classList.remove("is-copied"), 1400);
        });
        return;
      }
      const link = event.target.closest("a[href^='#term-']");
      if (link) {
        event.preventDefault();
        goTo(link.getAttribute("href").slice(1));
      }
    });

    tocList.addEventListener("click", (event) => {
      const link = event.target.closest("a[href^='#term-']");
      if (!link) return;
      event.preventDefault();
      goTo(link.getAttribute("href").slice(1));
    });

    let scheduled = false;
    window.addEventListener("scroll", () => {
      if (scheduled) return;
      scheduled = true;
      requestAnimationFrame(() => {
        scheduled = false;
        updateActive();
      });
    }, { passive: true });

    counts();
    const params = new URLSearchParams(location.search);
    if (params.get("term")) { query = params.get("term"); search.value = query; }
    const requested = params.get("category");
    if (requested && chips.some((chip) => chip.dataset.category === requested)) setCategory(requested);
    if (params.get("sort") === "az") setAlpha(true);
    apply();

    function openHash() {
      if (location.hash.length < 2) return;
      const card = resolve(location.hash);
      if (!card) return;
      if (card.hidden) {
        query = "";
        search.value = "";
        setCategory("all");
        apply();
        syncUrl();
      }
      window.setTimeout(() => goTo(card.id), 60);
    }

    openHash();
    window.addEventListener("hashchange", openHash);
  })();
</script>
